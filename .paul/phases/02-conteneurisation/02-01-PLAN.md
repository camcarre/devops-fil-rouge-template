<task type="auto">
  <name>Task 2: Créer les 4 modèles SQLAlchemy</name>

    src/models/user.py — table "users" :
    - id : Integer, PK, autoincrement
    - username : String(50), unique, not null
    - email : String(255), unique, not null
    - password_hash : String(255), not null
    - created_at : DateTime, default=datetime.utcnow

    src/models/category.py — table "categories" :
    - id : Integer, PK, autoincrement
    - name : String(100), unique, not null
    - description : Text, nullable
    - created_at : DateTime, default=datetime.utcnow

    src/models/topic.py — table "topics" :
    - id : Integer, PK, autoincrement
    - title : String(255), not null
    - category_id : Integer, ForeignKey("categories.id"), not null
    - user_id : Integer, ForeignKey("users.id"), not null
    - created_at : DateTime, default=datetime.utcnow
    - Relations : category = relationship("Category"), author = relationship("User")

    src/models/post.py — table "posts" :
    - id : Integer, PK, autoincrement
    - content : Text, not null
    - topic_id : Integer, ForeignKey("topics.id"), not null
    - user_id : Integer, ForeignKey("users.id"), not null
    - created_at : DateTime, default=datetime.utcnow
    - Relations : topic = relationship("Topic"), author = relationship("User")

    src/models/__init__.py : importer les 4 modèles (requis pour qu'Alembic les détecte via Base.metadata).
